/* =====================================================
   Meridian Diagnostics – api.js
   Loads the test catalogue from an API endpoint and shows
   it in a table.

   Uses: fetch(), async, await, response.json(),
         try, catch, finally,
         createElement(), appendChild(), textContent
   ===================================================== */

/* Change this to a real URL when you have a server, e.g.
   "https://api.example.com/v1/tests". The JSON must look like
   { "tests": [ { name, category, sample, fastingHours, reportHours, price } ] } */
const API_URL = "tests.json";

const statusBox   = document.getElementById("api-status");
const summaryText = document.getElementById("api-summary");
const tableWrap   = document.getElementById("table-wrap");
const tableBody   = document.getElementById("tests-body");
const reloadBtn   = document.getElementById("reload-btn");

/* ---------- Helpers to show loading / error states ---------- */
function showLoading() {
  statusBox.innerHTML = "";
  statusBox.className = "api-status loading";

  const spinner = document.createElement("span");
  spinner.className = "spinner";

  const text = document.createElement("span");
  text.textContent = "Loading tests…";

  statusBox.appendChild(spinner);
  statusBox.appendChild(text);

  tableWrap.hidden = true;
  summaryText.textContent = "Fetching data from the server…";
  reloadBtn.disabled = true;
}

function showError(message) {
  statusBox.innerHTML = "";
  statusBox.className = "api-status error";

  const title = document.createElement("strong");
  title.textContent = "We could not load the test list.";

  const detail = document.createElement("p");
  detail.textContent = message;

  statusBox.appendChild(title);
  statusBox.appendChild(detail);

  tableWrap.hidden = true;
  summaryText.textContent = "No data loaded.";
}

function clearStatus() {
  statusBox.innerHTML = "";
  statusBox.className = "api-status";
}

/* ---------- Build one table row from one test object ---------- */
function createCell(text, className) {
  const td = document.createElement("td");
  td.textContent = text;
  if (className) {
    td.className = className;
  }
  return td;
}

function createRow(test) {
  const tr = document.createElement("tr");

  tr.appendChild(createCell(test.name, "test-name"));
  tr.appendChild(createCell(test.category));
  tr.appendChild(createCell(test.sample));
  tr.appendChild(createCell(test.fastingHours > 0 ? test.fastingHours + " hrs" : "Not needed"));
  tr.appendChild(createCell(test.reportHours + " hrs"));
  tr.appendChild(createCell("₹" + Number(test.price).toLocaleString("en-IN"), "price-cell"));

  /* Last column: a button that fills in the booking form */
  const actionCell = document.createElement("td");
  const bookButton = document.createElement("button");
  bookButton.type = "button";
  bookButton.className = "btn btn-small";
  bookButton.textContent = "Book";
  bookButton.addEventListener("click", function () {
    selectTestInForm(test.name);
  });
  actionCell.appendChild(bookButton);
  tr.appendChild(actionCell);

  return tr;
}

/* Choose the test in the booking form and jump to it */
function selectTestInForm(testName) {
  const select = document.getElementById("test");
  let option = Array.from(select.options).find(function (opt) {
    return opt.value === testName || opt.textContent === testName;
  });

  /* The catalogue has more tests than the form lists, so add it if missing */
  if (!option) {
    option = document.createElement("option");
    option.textContent = testName;
    select.appendChild(option);
  }

  select.value = option.value || option.textContent;
  select.dispatchEvent(new Event("change", { bubbles: true }));
  document.getElementById("booking").scrollIntoView({ behavior: "smooth" });
  document.getElementById("name").focus({ preventScroll: true });
}

/* ---------- Fill the table with the received data ---------- */
function renderTable(tests) {
  tableBody.innerHTML = "";

  tests.forEach(function (test) {
    tableBody.appendChild(createRow(test));
  });

  tableWrap.hidden = false;
}

/* ---------- Main function: fetch, check, show ---------- */
async function loadTests() {
  showLoading();

  try {
    /* Short pause so the loading spinner is visible when testing locally.
       Delete this line when using a real server. */
    await new Promise(function (resolve) { setTimeout(resolve, 700); });

    const response = await fetch(API_URL);

    /* fetch() only fails on network problems, so check the status ourselves */
    if (!response.ok) {
      throw new Error("The server replied with status " + response.status + ".");
    }

    const data = await response.json();

    if (!Array.isArray(data.tests) || data.tests.length === 0) {
      throw new Error("The server sent no tests.");
    }

    clearStatus();
    renderTable(data.tests);
    summaryText.textContent =
      data.tests.length + " tests available. Last updated " + data.updated + ".";

  } catch (error) {
    let message = error.message;

    /* A network failure (or opening index.html straight from a folder) */
    if (error instanceof TypeError) {
      message =
        "Could not reach the server. If you opened index.html directly from a folder, " +
        "run the page with a local server (for example, VS Code Live Server) and try again.";
    }
    /* Invalid JSON from the server */
    if (error instanceof SyntaxError) {
      message = "The server sent data in an unexpected format.";
    }

    showError(message);

  } finally {
    /* Runs whether the request worked or failed */
    reloadBtn.disabled = false;
  }
}
